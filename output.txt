1073:     // === Input Manager Class ===
1074:     class InputManager {
1075:         constructor(targetElement) {
1076:             this.el = targetElement;
1077:             this.lookPointerId = null;
1078:             this.lastX = 0; this.lastY = 0;
1079:             this.lookDX = 0; this.lookDY = 0;
1080:             this.LOOK_SENSITIVITY = 0.003;
1081:             this._bind();
1082:         }
1083: 
1084:         consumeLookDelta() {
1085:             const dx = this.lookDX; const dy = this.lookDY;
1086:             this.lookDX = 0; this.lookDY = 0;
1087:             return { dx, dy };
1088:         }
1089: 
1090:         _bind() {
1091:             this.el.style.touchAction = 'none'; // ブラウザのジェスチャーを無効化
1092:             this.el.addEventListener('mousedown', this._onPointerDown);
1093:             this.el.addEventListener('mousemove', this._onPointerMove);
1094:             this.el.addEventListener('mouseup', this._onPointerUp);
1095:             this.el.addEventListener('mouseleave', this._onPointerUp);
1096: 
1097:             // passive: false でピンチズーム等を確実にブロック
1098:             this.el.addEventListener('touchstart', this._onPointerDown, { passive: false });
1099:             this.el.addEventListener('touchmove', this._onPointerMove, { passive: false });
1100:             this.el.addEventListener('touchend', this._onPointerUp);
1101:             this.el.addEventListener('touchcancel', this._onPointerUp);
1102:         }
1103: 
1104:         _onPointerDown = (e) => {
1105:             // UI操作（D-Pad等）はInputManagerとしては無視
1106:             // ※D-Pad側のイベントリスナーで処理されるため
1107:             if (e.target.closest('#sg-dpad') || e.target.closest('.dpad-btn')) return;
1108: 
1109:             if (e.type === 'mousedown') {
1110:                 this.lookPointerId = 'mouse';
1111:                 this.lastX = e.clientX;
1112:                 this.lastY = e.clientY;
1113:                 return; // ここでreturnしないとtouchstartの判定に入ってしまう可能性があるため修正
1114:             }
1115: 
1116:             if (e.type === 'touchstart') {
1117:                 // 【修正ポイント】
1118:                 // 「指が1本だけの時(e.touches.length === 1)」という制限を撤廃します。
1119:                 // 代わりに、「新しい指が視点用として有効か」を個別に判定します。
1120: 
1121:                 for (const t of e.changedTouches) {
1122:                     // すでに視点操作中の指(lookPointerId)が存在する場合、
1123:                     // 新たな指は「2本目の視点指（ピンチ操作）」とみなして無視します。
1124:                     if (this.lookPointerId !== null) continue;
1125: 
1126:                     // D-Padエリア外の指であれば、これを「視点操作用の指」として登録します。
1127:                     // (移動用の指があっても、lookPointerIdがnullなら登録されます)
1128:                     this.lookPointerId = t.identifier;
1129:                     this.lastX = t.clientX;
1130:                     this.lastY = t.clientY;
1131: 
1132:                     // 視点指が決まったらループを抜けます
1133:                     break;
1134:                 }
1135: 
1136:                 // 視点指として登録された場合のみ、デフォルト動作を防ぎます
1137:                 if (this.lookPointerId !== null) {
1138:                     e.preventDefault();
1139:                 }
1140:             }
1141:         };
1142: 
1143:         _onPointerMove = (e) => {
1144:             if (this.lookPointerId === null) return;
1145:             let x, y;
1146: 
1147:             if (e.type === 'mousemove' && this.lookPointerId === 'mouse') {
1148:                 x = e.clientX; y = e.clientY;
1149:             } else if (e.type === 'touchmove') {
1150:                 const t = [...e.changedTouches].find(t => t.identifier === this.lookPointerId);
1151:                 if (!t) return;
1152:                 x = t.clientX; y = t.clientY;
1153:                 e.preventDefault();
1154:             } else { return; }
1155: 
1156:             this.lookDX += (x - this.lastX) * this.LOOK_SENSITIVITY;
1157:             this.lookDY += (y - this.lastY) * this.LOOK_SENSITIVITY;
1158:             this.lastX = x; this.lastY = y;
1159:         };
1160: 
1161:         _onPointerUp = (e) => {
1162:             if (this.lookPointerId === 'mouse' && (e.type === 'mouseup' || e.type === 'mouseleave')) {
1163:                 this.lookPointerId = null;
1164:             } else if (e.type.startsWith('touch')) {
1165:                 for (const t of e.changedTouches) {
1166:                     if (t.identifier === this.lookPointerId) {
1167:                         this.lookPointerId = null; break;
1168:                     }
1169:                 }
1170:             }
1171:         };
1172: 
1173:         dispose() {
1174:             this.el.removeEventListener('mousedown', this._onPointerDown);
1175:             this.el.removeEventListener('mousemove', this._onPointerMove);
1176:             this.el.removeEventListener('mouseup', this._onPointerUp);
1177:             this.el.removeEventListener('mouseleave', this._onPointerUp);
1178:             this.el.removeEventListener('touchstart', this._onPointerDown);
1179:             this.el.removeEventListener('touchmove', this._onPointerMove);
1180:             this.el.removeEventListener('touchend', this._onPointerUp);
1181:             this.el.removeEventListener('touchcancel', this._onPointerUp);
1182:         }
1183:     }
